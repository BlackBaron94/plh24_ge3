CREATE TABLE CATEGORY (
  category_id INT GENERATED ALWAYS AS IDENTITY (START WITH 1, INCREMENT BY 1) PRIMARY KEY,
  category_name VARCHAR(100) NOT NULL,
  CONSTRAINT UNIQUE_CATEGORY_NAME UNIQUE (category_name)
);


CREATE TABLE ARTICLE (
  article_id INT GENERATED ALWAYS AS IDENTITY (START WITH 1, INCREMENT BY 1) PRIMARY KEY,
  title VARCHAR(300) NOT NULL,
  rating SMALLINT CHECK (rating BETWEEN 1 AND 5),
  saved_at TIMESTAMP NOT NULL,
  category_id INT NOT NULL,
  comments VARCHAR(2000),
  CONSTRAINT UNIQUE_TITLE UNIQUE (title),
  CONSTRAINT FK_CATEGORY_ID FOREIGN KEY (category_id)
    REFERENCES CATEGORY(category_id)
);

CREATE TABLE SEARCH_LOG (
  search_id INT GENERATED ALWAYS AS IDENTITY (START WITH 1, INCREMENT BY 1) PRIMARY KEY,
  search_term VARCHAR(200) NOT NULL,
  searched_at TIMESTAMP NOT NULL
);
